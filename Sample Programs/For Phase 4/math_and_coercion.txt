dream {
coercion_math.txt – Test automatic type conversion (coercion) during
arithmetic operations and assignments. }

activate {
    dream: ----- setup values -----
    pulse   p = 10;
    spark   s = 3.5f;
    stream  d = 2.0;
    synapse flag = true;

    dream: ----- mixed-type addition -----
    spark mix1 = p + s;                // pulse + spark -> spark
    express("10(pulse) + 3.5(spark) = spark: " + transcribe(mix1));

    stream mix2 = s + d;               // spark + stream -> stream
    express("3.5(spark) + 2.0(stream) = stream: " + transcribe(mix2));

    stream mix3 = p + d;               // pulse + stream -> stream
    express("10(pulse) + 2.0(stream) = stream: " + transcribe(mix3));

    dream: ----- compound assignment with coercion -----
    p += s;                            // pulse gets float value -> auto coerced
    express("p after += 3.5 : " + transcribe(p));

    dream: ----- boolean to numeric (true=1, false=0) -----
    pulse plus_true = flag + 5;
    express("true + 5 = " + transcribe(plus_true));

    pulse plus_false = (false) + 5;
    express("false + 5 = " + transcribe(plus_false));

    dream: ----- string concatenation (both thought) -----
    thought hello = "Hello";
    thought world = "World";
    thought greeting = hello + " " + world;
    express("Concatenation: " + greeting);

    dream: ----- relational coercion -----
    stimulate (p > s) {
        express("p > s is true after coercion.");
    } inhibit {
        express("p > s is false.");
    }

    dream: ----- auto‑declaration on assignment (optional feature) -----
    autoVar = "This is born on assignment!";
    express(autoVar);
}
