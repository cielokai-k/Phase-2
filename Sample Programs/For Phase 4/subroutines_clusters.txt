action pulse getWeightedSum(cluster pulse data[], cluster pulse weights[], pulse size) {
    pulse totalSum = 0;
    pulse i;
    
    dream: Echo loop with pre-declared variable
    echo (i = 0; i < size; i++) {
        totalSum = totalSum + (data[i] * weights[i]);
    }
    
    recall totalSum;
}
