dream {
nested_control_flows.txt – Nested stimulate, cycle inside echo,
react, evaluate, and loop control statements (flow, dormant). }

activate {
    dream: ----- nested stimulate (if‑else) -----
    pulse age = 22;
    synapse has_license = true;

    stimulate (age >= 18) {
        express("Adult.");
        stimulate (has_license) {
            express("  Can drive.");
        } inhibit {
            express("  Cannot drive (no license).");
        }
    } inhibit {
        express("Minor.");
        stimulate (has_license) {
            express("  Has license but underage.");
        } inhibit {
            express("  No license, too young.");
        }
    }

    dream: ----- cycle loop inside echo loop -----
    pulse i = 1;
    echo (i = 1; i <= 3; i++) {
        express("Outer echo iteration: " + transcribe(i));
        pulse j = 1;
        cycle (j <= i) {
            express("  Cycle inner: " + transcribe(j));
            j++;
        }
    }

    dream: ----- react (do‑while) nested with cycle -----
    pulse row = 0;
    react {
        express("React row: " + transcribe(row));
        pulse col = 0;
        cycle (col < row) {
            express("  cycle col: " + transcribe(col));
            col++;
        }
        row++;
    } cycle (row < 3);

    dream: ----- evaluate (switch) inside a loop -----
    pulse k = 1;
    cycle (k <= 3) {
        evaluate (k) {
            path 1: express("  -> one (odd)"); dormant;
            path 2: express("  -> two (even)"); dormant;
            path 3: express("  -> three (odd)"); dormant;
            base: express("  -> other");
        }
        k++;
    }

    dream: ----- flow (continue) inside nested stimulate -----
    pulse m = 0;
    cycle (m < 5) {
        m++;
        stimulate (m == 2) {
            flow;   dream: skip printing 2
        }
        stimulate (m == 4) {
            express("Almost done...");
        }
        express("m = " + transcribe(m));
    }

    dream: ----- dormant (break) in nested loops -----
    pulse found = 0;
    pulse candidate = 2;
    cycle (candidate <= 10) {
        pulse divisor = 2;
        synapse is_prime = true;
        cycle (divisor < candidate) {
            stimulate (candidate % divisor == 0) {
                is_prime = false;
                dormant;   dream: break inner cycle
            }
            divisor++;
        }
        stimulate (is_prime) {
            express(transcribe(candidate) + " is prime.");
            found++;
            stimulate (found == 2) {    
                dormant;   dream: break outer cycle after 2 primes
            }
        }
        candidate++;
    }
    express("Found " + transcribe(found) + " primes.");
}
