activate {
    dream {
        OPTIONAL FEATURES DEMONSTRATION
        This covers instinct, flow, 2D clusters, multiple declarations, 
        and Explicit Type Conversion (transcribe).
    }

    dream {
        1. Constant Data Type (instinct)
        The value of 'limit' cannot be changed after this line.
        It acts as a fixed boundary for our loops.
    }

    instinct pulse limit = 3;

    dream {
        2. Multiple Variable Declarations
        Declaring multiple variables of the same type in one line.
    }

    pulse row, col, sum;

    dream {
        3. 2D Arrays (Cluster)
        Initializing a 3x3 matrix using nested curly braces.
    }

    pulse cluster matrix[3][3] = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };

    dream {
        4. Explicit Type Conversion (transcribe)
        Cerebra does not allow implicit casting during concatenation.
        We use 'transcribe()' to convert the integer (pulse) 'row' 
        into a string (thought) before adding it to 'prefix'.
    }

    thought prefix = "Processing Row ";
    
    echo (row = 0; row < limit; row++) {
        
        sum = 0;
        
        dream: Concatenating String + Transcribed Integer + String
        express(prefix + transcribe(row) + ":");

        echo (col = 0; col < limit; col++) {
            
            pulse currentValue = matrix[row][col];

            dream {
                5. Loop Control (flow)
                If the value is even, 'flow' skips the rest of the loop block.
            }

            stimulate (currentValue % 2 == 0) {
                flow; 
            }

            sum = sum + currentValue;
        }

        dream: Another usage of transcribe for the final output
        express("Sum of odds: " + transcribe(sum));
    }
}
