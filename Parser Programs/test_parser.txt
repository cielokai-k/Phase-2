dream: functions

action void printSeparator() {
    express("---");
}

action pulse add(pulse a, pulse b) {
    recall a + b;
}

action stream average(stream x, stream y) {
    recall (x + y) / 2.0;
}

action synapse isEven(pulse n) {
    recall n % 2 == 0;
}


action void printFirst(cluster pulse arr[]) {
    express(arr[0]);
}

    spark   pi      = 3.14f;
