dream: ---- helper function: no parameters, void return ----
action void printSeparator() {
    express("---");
}

dream: ---- function with parameters and a return value ----
action pulse add(pulse a, pulse b) {
    recall a + b;
}

dream: ---- function returning a stream ----
action stream average(stream x, stream y) {
    recall (x + y) / 2.0;
}

dream: ---- function with a synapse return type ----
action synapse isEven(pulse n) {
    recall n % 2 == 0;
}

dream { ---- function that takes a cluster (1D array) parameter ----
  CFG rule: CLUSTER <data_type> IDENTIFIER L_BRACKET R_BRACKET
  So the keyword 'cluster' must come FIRST, then the data type.
  Correct:  cluster pulse arr[]
  Wrong:    pulse cluster arr[]   (data type before cluster keyword)
}
action void printFirst(cluster pulse arr[]) {
    express(arr[0]);
}

activate {

    dream: --- primitive declarations with initial values ---
    pulse   count   = 0;
    spark   pi      = 3.14f;
    stream  ratio   = 9.81;
    thought msg     = "hello";
    neuron  letter  = 'A';
    synapse flag    = true;

    dream: --- declaration without initial value ---
    pulse unset;

    dream: --- multiple variables on one line ---
    pulse x = 1, y = 2, z;

    dream: --- instinct (constant) ---
    instinct pulse MAX = 100;
    instinct stream GRAVITY = 9.81;

    dream: --- 1D cluster declaration and initialisation ---
    pulse cluster scores[5] = {10, 20, 30, 40, 50};

    dream: --- 2D cluster declaration and initialisation ---
    pulse cluster grid[2][3] = {{1, 2, 3}, {4, 5, 6}};

    dream: --- cluster without initialiser ---
    stream cluster temps[7];

    dream: --- assignments with every compound operator ---
    count = 10;
    count += 5;
    count -= 2;
    count *= 3;
    count /= 2;
    count %= 4;

    dream: --- array element assignment ---
    scores[0] = 99;
    grid[1][2] = 42;

    dream: --- express with arithmetic expression ---
    express(count + MAX);

    dream: --- express with string concatenation ---
    express(msg + " world");

    dream: --- express using transcribe ---
    express("count is: " + transcribe(count));

    dream: --- express using length ---
    express(length(msg));

    dream: --- sense (input) ---
    sense pulse count;
    sense thought msg;

    dream: --- function calls as statements ---
    printSeparator();

    dream: --- function call result assigned to variable ---
    pulse total = add(x, y);
    stream avg  = average(1.0, 2.0);
    synapse even = isEven(count);

    dream: --- stimulate with inhibit (if / else) ---
    stimulate (count > 0) {
        express("positive");
    } inhibit {
        express("non-positive");
    }

    dream: --- nested stimulate ---
    stimulate (flag) {
        stimulate (count == 0) {
            express("flag true, count zero");
        }
    }

    stimulate (x != y) {
        express("x and y differ");
    }

    dream: --- cycle with flow and dormant ---
    pulse i = 0;
    cycle (i < 20) {
        i++;
        stimulate (i == 5) {
            flow;
        }
        stimulate (i == 15) {
            dormant;
        }
        express(transcribe(i));
    }

    dream: --- react (do-while) ---
    pulse retry = 0;
    react {
        retry++;
        express("retrying...");
    } cycle (retry < 3);

    dream: --- echo with declaration init and ++ update ---

        express(transcribe(j));
    }

    dream: --- echo with assignment-style init (not a declaration) ---
    pulse m = 0;
    echo (m = 0; m < 3; m++) {
        express("m = " + transcribe(m));

            express("one");
            dormant;
